// Audio synthétique autonome
let soundOn = true;
const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
function playTone(freq, type, dur, vol=0.15) {
  if (!soundOn) return;
  if (audioCtx.state === 'suspended') audioCtx.resume();
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type;
  osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
  osc.frequency.exponentialRampToValueAtTime(20, audioCtx.currentTime + dur);
  gain.gain.setValueAtTime(vol, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
  osc.connect(gain); gain.connect(audioCtx.destination);
  osc.start(); osc.stop(audioCtx.currentTime + dur);
}
function toggleSound() {
  soundOn = !soundOn;
  document.getElementById('sound-btn').textContent = soundOn ? '🔊' : '🔇';
}

  const m = document.getElementById('msg');

}

  let dx = t.clientX - (r.left + 60), dy = t.clientY - (r.top + 60);

  joyVec.y = (Math.sin(ang) * dist) / 40;

// Actions de gameplay
function shoot() {
  if (player.dead) return;
  player.recoil = 10;
  const v = player.inVehicle;
  const isTank = v && v.type === 'tank';
  playTone(isTank ? 90 : 320, isTank ? 'sawtooth' : 'square', isTank ? 0.35 : 0.12, 0.2);
  const ox = v ? v.x : player.x, oy = v ? v.y : player.y, oz = v ? v.z + 15 : 18;
  const ang = v ? v.angle : player.angle;
  bullets.push({ x: ox + Math.cos(ang)*30, y: oy + Math.sin(ang)*30, z: oz, vx: Math.cos(ang)*16, vy: Math.sin(ang)*16, life: 55, explosive: isTank, owner: 'player' });
  spawnParticles(ox + Math.cos(ang)*28, oy + Math.sin(ang)*28, oz, '#fde047', 5);
}

function interact() {
  if (player.dead) return;
  if (player.inVehicle) {
    player.x = player.inVehicle.x + 40; player.y = player.inVehicle.y; player.z = 0;
    notify('Sorti de : ' + player.inVehicle.name);
    player.inVehicle = null;
    return;
  }
  for (const v of vehicles) {
    if (Math.hypot(player.x - v.x, player.y - v.y) < 70) {
      player.inVehicle = v;
      notify('Véhicule : ' + v.name + (v.type === 'plane' ? ' (Accélère pour décoller !)' : ''));
      return;
    }
  }
  notify('Aucun véhicule à proximité');
}

function recruit() {
  for (const n of npcs) {
    if (!n.ally && !n.cop && Math.hypot(player.x - n.x, player.y - n.y) < 90) {
      if (player.cash >= 100) {
        player.cash -= 100; n.ally = true; n.color = '#10b981';
        notify('Membre recruté dans ton gang (-100$) !');
        playTone(600, 'sine', 0.2);
      } else notify('Il faut 100$ pour recruter !');
      return;
    }
  }
  notify('Approche-toi d\'un civil violet pour le recruter');
}

function robBank() {
  for (const b of buildings) {
    if (b.isBank && player.x > b.x - 40 && player.x < b.x + b.w + 40 && player.y > b.y - 40 && player.y < b.y + b.d + 40) {
      player.cash += 1500;
      player.wanted = Math.min(5, player.wanted + 2);
      playTone(880, 'triangle', 0.4);
      spawnParticles(player.x, player.y, 30, '#fde047', 25);
      notify('💰 BRAQUAGE RÉUSSI ! +1500$ (Police alertée !)');
      return;
    }
  }
  notify('Approche-toi d\'une Banque dorée pour braquer');
}

function spawnParticles(x, y, z, color, count) {
  for (let i = 0; i < count; i++) {
    particles.push({
      x, y, z,
      vx: (Math.random() - 0.5) * 8, vy: (Math.random() - 0.5) * 8, vz: Math.random() * 6 + 2,
      life: 25 + Math.random() * 15, color
    });
  }
}

  if (player.dead) return;
  player.dead = true;
  playTone(55, 'sine', 2.8, 0.45); // Bruit sourd

// Multijoueur WebRTC P2P sans serveur externe
function openP2P() { document.getElementById('modal').style.display = 'flex'; }
function closeP2P() { document.getElementById('modal').style.display = 'none'; }
function setupPeer() {
  peer = new RTCPeerConnection({ iceServers: [] });
  peer.onicecandidate = e => { if (!e.candidate) document.getElementById('p2p-out').value = JSON.stringify(peer.localDescription); };
  peer.ondatachannel = e => { dataChan = e.channel; bindChan(); };
}
function bindChan() {
  dataChan.onopen = () => { notify('🌐 Joueur connecté en P2P !'); closeP2P(); };
  dataChan.onmessage = e => { remotePlayer = JSON.parse(e.data); };
}
async function createOffer() {
  setupPeer();
  dataChan = peer.createDataChannel('gta');
  bindChan();
  const offer = await peer.createOffer();
  await peer.setLocalDescription(offer);
}
async function createAnswer() {
  setupPeer();
  const raw = document.getElementById('p2p-in').value.trim();
  if (!raw) return;
  await peer.setRemoteDescription(JSON.parse(raw));
  const ans = await peer.createAnswer();
  await peer.setLocalDescription(ans);
}
async function applyRemote() {
  const raw = document.getElementById('p2p-in').value.trim();
  if (peer && raw) await peer.setRemoteDescription(JSON.parse(raw));
}

// Moteur de projection 3D isométrique/perspective extrudée
function proj(x, y, z, camX, camY) {
  const rx = x - camX, ry = y - camY;
  return { sx: W / 2 + rx, sy: H / 2 + ry * 0.72 - z };
}

function drawBox3D(x, y, z, w, d, h, topCol, sideCol, neonCol, camX, camY) {
  const p1 = proj(x, y, z, camX, camY);
  const p2 = proj(x + w, y, z, camX, camY);
  const p3 = proj(x + w, y + d, z, camX, camY);
  const p4 = proj(x, y + d, z, camX, camY);
  const t1 = proj(x, y, z + h, camX, camY);
  const t2 = proj(x + w, y, z + h, camX, camY);
  const t3 = proj(x + w, y + d, z + h, camX, camY);
  const t4 = proj(x, y + d, z + h, camX, camY);

  // Face avant (Sud)
  ctx.fillStyle = sideCol;
  ctx.beginPath(); ctx.moveTo(p4.sx, p4.sy); ctx.lineTo(p3.sx, p3.sy); ctx.lineTo(t3.sx, t3.sy); ctx.lineTo(t4.sx, t4.sy); ctx.closePath(); ctx.fill();
  // Face droite (Est)
  ctx.fillStyle = '#0f172a';
  ctx.beginPath(); ctx.moveTo(p3.sx, p3.sy); ctx.lineTo(p2.sx, p2.sy); ctx.lineTo(t2.sx, t2.sy); ctx.lineTo(t3.sx, t3.sy); ctx.closePath(); ctx.fill();
  // Toit
  ctx.fillStyle = topCol;
  ctx.beginPath(); ctx.moveTo(t1.sx, t1.sy); ctx.lineTo(t2.sx, t2.sy); ctx.lineTo(t3.sx, t3.sy); ctx.lineTo(t4.sx, t4.sy); ctx.closePath(); ctx.fill();
  if (neonCol) {
    ctx.strokeStyle = neonCol; ctx.lineWidth = 2; ctx.stroke();
  }
}

// Boucle principale
function loop() {
  let mx = joyVec.x + (keys['d'] || keys['arrowright'] ? 1 : 0) - (keys['a'] || keys['q'] || keys['arrowleft'] ? 1 : 0);
  let my = joyVec.y + (keys['s'] || keys['arrowdown'] ? 1 : 0) - (keys['w'] || keys['z'] || keys['arrowup'] ? 1 : 0);
  const mag = Math.hypot(mx, my);
  if (mag > 1) { mx /= mag; my /= mag; }

  if (!player.dead) {
    if (player.inVehicle) {
      const v = player.inVehicle;
      if (mag > 0.1) {
        v.angle = Math.atan2(my, mx);
        const maxSpd = v.type === 'plane' ? 12 : (v.type === 'car' ? 8.5 : 4.5);
        v.speed = Math.min(maxSpd, v.speed + 0.25);
      } else {
        v.speed *= 0.94;
      }
      v.x += Math.cos(v.angle) * v.speed;
      v.y += Math.sin(v.angle) * v.speed;
      if (v.type === 'plane') {
        v.z = v.speed > 6 ? Math.min(180, v.z + 2) : Math.max(0, v.z - 2);
      }
      player.x = v.x; player.y = v.y; player.z = v.z; player.angle = v.angle;
    } else if (mag > 0.1) {
      player.angle = Math.atan2(my, mx);
      player.x += mx * 4.2;
      player.y += my * 4.2;
    }
  }
  if (player.recoil > 0) player.recoil *= 0.8;

  // Mise à jour balles & explosions
  for (let i = bullets.length - 1; i >= 0; i--) {
    const b = bullets[i];
    b.x += b.vx; b.y += b.vy; b.life--;
    for (let j = npcs.length - 1; j >= 0; j--) {
      const n = npcs[j];
      if (!n.ally && Math.hypot(b.x - n.x, b.y - n.y) < 24) {
        n.hp -= b.explosive ? 100 : 25;
        b.life = 0;
        spawnParticles(n.x, n.y, 15, '#ff3b5c', b.explosive ? 22 : 6);
        if (n.hp <= 0) {
          npcs.splice(j, 1);
          player.cash += 75;
          if (n.cop) player.wanted = Math.min(5, player.wanted + 1);
          playTone(140, 'sawtooth', 0.25);
        }
      }
    }
    if (b.life <= 0) bullets.splice(i, 1);
  }

  // IA PNJ & Police
  for (const n of npcs) {
    if (n.ally) {
      const d = Math.hypot(player.x - n.x, player.y - n.y);
      if (d > 75) {
        const a = Math.atan2(player.y - n.y, player.x - n.x);
        n.x += Math.cos(a) * 3.8; n.y += Math.sin(a) * 3.8;
      }
    } else if (n.cop && player.wanted > 0 && !player.dead) {
      const a = Math.atan2(player.y - n.y, player.x - n.x);
      n.x += Math.cos(a) * 3.1; n.y += Math.sin(a) * 3.1;
      if (Math.hypot(player.x - n.x, player.y - n.y) < 35 && player.z < 20) {
        player.hp -= 0.4;
        if (player.hp <= 0) { player.hp = 0; triggerDeath(); }
      }
    } else {
      n.x += Math.cos(n.angle) * 1.2; n.y += Math.sin(n.angle) * 1.2;
      if (Math.random() < 0.02) n.angle += (Math.random() - 0.5) * 2;
    }
  }

  // Particules
  for (let i = particles.length - 1; i >= 0; i--) {
    const p = particles[i];
    p.x += p.vx; p.y += p.vy; p.z = Math.max(0, p.z + p.vz); p.vz -= 0.35; p.life--;
    if (p.life <= 0) particles.splice(i, 1);
  }

  // Envoi position P2P
  if (dataChan && dataChan.readyState === 'open') {
    dataChan.send(JSON.stringify({ x: player.x, y: player.y, z: player.z, angle: player.angle }));
  }

  // Rendu
  ctx.fillStyle = '#070b19';
  ctx.fillRect(0, 0, W, H);
  const camX = player.x, camY = player.y - player.z * 0.5;

  // Grille des routes néon
  ctx.strokeStyle = 'rgba(0, 242, 254, 0.14)';
  ctx.lineWidth = 2;
  const startX = Math.floor((camX - W) / 180) * 180;
  const startY = Math.floor((camY - H) / 180) * 180;
  for (let gx = startX; gx < camX + W; gx += 180) {
    const pA = proj(gx, camY - H, 0, camX, camY), pB = proj(gx, camY + H, 0, camX, camY);
    ctx.beginPath(); ctx.moveTo(pA.sx, pA.sy); ctx.lineTo(pB.sx, pB.sy); ctx.stroke();
  }
  for (let gy = startY; gy < camY + H; gy += 180) {
    const pA = proj(camX - W, gy, 0, camX, camY), pB = proj(camX + W, gy, 0, camX, camY);
    ctx.beginPath(); ctx.moveTo(pA.sx, pA.sy); ctx.lineTo(pB.sx, pB.sy); ctx.stroke();
  }

  // Tri en profondeur (Painter's Algorithm 3D)
  const renderList = [];
  for (const b of buildings) renderList.push({ y: b.y + b.d, fn: () => drawBox3D(b.x, b.y, 0, b.w, b.d, b.h, b.color, '#111827', b.neon, camX, camY) });
  for (const v of vehicles) renderList.push({ y: v.y + v.d, fn: () => {
    if (v.z > 2) { const sh = proj(v.x, v.y, 0, camX, camY); ctx.fillStyle = 'rgba(0,0,0,0.45)'; ctx.beginPath(); ctx.ellipse(sh.sx, sh.sy, 26, 14, 0, 0, Math.PI*2); ctx.fill(); }
    drawBox3D(v.x - v.w/2, v.y - v.d/2, v.z, v.w, v.d, v.h, v.color, '#1e293b', '#fff', camX, camY);
  }});
  for (const n of npcs) renderList.push({ y: n.y, fn: () => drawBox3D(n.x - 10, n.y - 10, 0, 20, 20, 34, n.color, '#0f172a', n.ally ? '#fde047' : null, camX, camY) });
  if (!player.inVehicle) renderList.push({ y: player.y, fn: () => {
    drawBox3D(player.x - 11, player.y - 11, player.z, 22, 22, 36, '#00f2fe', '#0284c7', '#fff', camX, camY);
    const p = proj(player.x, player.y, player.z + 24, camX, camY);
    const gunLen = 22 - player.recoil;
    ctx.strokeStyle = '#fde047'; ctx.lineWidth = 4;
    ctx.beginPath(); ctx.moveTo(p.sx, p.sy); ctx.lineTo(p.sx + Math.cos(player.angle)*gunLen, p.sy + Math.sin(player.angle)*gunLen*0.72); ctx.stroke();
  }});
  if (remotePlayer) renderList.push({ y: remotePlayer.y, fn: () => drawBox3D(remotePlayer.x - 11, remotePlayer.y - 11, remotePlayer.z, 22, 22, 36, '#ec4899', '#be185d', '#fde047', camX, camY) });

  renderList.sort((a, b) => a.y - b.y).forEach(item => item.fn());

  // Balles & Particules
  for (const b of bullets) {
    const p = proj(b.x, b.y, b.z, camX, camY);
    ctx.fillStyle = b.explosive ? '#ff3b5c' : '#fde047';
    ctx.beginPath(); ctx.arc(p.sx, p.sy, b.explosive ? 6 : 3.5, 0, Math.PI * 2); ctx.fill();
  }
  for (const pt of particles) {
    const p = proj(pt.x, pt.y, pt.z, camX, camY);
    ctx.fillStyle = pt.color; ctx.fillRect(p.sx - 3, p.sy - 3, 6, 6);
  }

  // HUD
  document.getElementById('hp').textContent = Math.ceil(player.hp);
  document.getElementById('cash').textContent = player.cash;
  document.getElementById('wanted').textContent = '★'.repeat(player.wanted) || '0';
  document.getElementById('gang').textContent = npcs.filter(n => n.ally).length;

  requestAnimationFrame(loop);
}
